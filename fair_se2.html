/* 📸 カメラ制御（フォーマット指定・エラー対策完了版） */ async function startCamera(mode) { if (mode === 'product' && !customer) return alert("先に顧客を選択してください"); showPage('scanner-page'); document.getElementById('camera-guide').style.display = 'block'; if (html5QrCode) { try { await html5QrCode.stop(); } catch (e) {} html5QrCode = null; } document.getElementById("reader").innerHTML = ""; try { const devices = await Html5Qrcode.getCameras(); if (!devices || devices.length === 0) return alert("カメラが見つかりません"); const backCameras = devices.filter(d => d.label && (d.label.toLowerCase().includes('back') || d.label.toLowerCase().includes('rear')) ); const targetId = backCameras.length > 0 ? backCameras[backCameras.length - 1].id : devices[0].id; const isCustomerMode = (mode === 'customer'); // モードに応じたフォーマット絞り込み設定 let formats = null; if (typeof Html5QrcodeSupportedFormats !== 'undefined') { formats = isCustomerMode ? [ Html5QrcodeSupportedFormats.QR_CODE ] : [ Html5QrcodeSupportedFormats.EAN_13, Html5QrcodeSupportedFormats.EAN_8, Html5QrcodeSupportedFormats.CODE_128, Html5QrcodeSupportedFormats.UPC_A ]; } const qrboxSize = isCustomerMode ? { width: 250, height: 250 } : { width: 280, height: 120 }; html5QrCode = formats ? new Html5Qrcode("reader", { formatsToSupport: formats }) : new Html5Qrcode("reader"); await html5QrCode.start( targetId, { fps: 15, qrbox: qrboxSize, aspectRatio: 1.0, videoConstraints: { facingMode: "environment", width: { ideal: 1280 }, height: { ideal: 720 } } }, (text) => { html5QrCode.stop().then(() => { document.getElementById('camera-guide').style.display = 'none'; showPage('home'); if (isCustomerMode) fetchCustomer(text.trim()); else fetchProduct(text.trim()); }).catch(err => console.error("Stop error", err)); } ); } catch (err) { console.error(err); alert("カメラを起動できませんでした。ブラウザの権限を確認してください。"); showPage('home'); } }
